SHELL=/bin/bash
AAC=/root/aac
# --- AACapital IPO pipeline (times are IST; VM tz set by bootstrap) ---
# Kite token refresh — daily 08:00 (login works any day; keeps preflight green incl weekends)
0 8 * * *    cd $AAC && set -a && . ./.env && set +a && venv/bin/python _scripts/refresh_kite_token.py  >> logs/token.log 2>&1
# Listing-day tick capture — 09:14 Mon–Fri (runs until 15:35)
14 9 * * 1-5 cd $AAC && set -a && . ./.env && set +a && venv/bin/python _scripts/ipo_tick_capture.py    >> logs/ticks.log 2>&1
# Nightly pipeline — 18:30 daily (scrape -> regime -> candles -> levels -> gate)
30 18 * * *  cd $AAC && set -a && . ./.env && set +a && venv/bin/python _scripts/run_ipo_pipeline.py    >> logs/pipe.log  2>&1
# Weekly purge — Sunday 19:00
0 19 * * 0   cd $AAC && set -a && . ./.env && set +a && venv/bin/python _scripts/run_ipo_pipeline.py --weekly >> logs/purge.log 2>&1
